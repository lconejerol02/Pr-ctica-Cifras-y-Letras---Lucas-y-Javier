        } while (posicion < 1 || posicion > 6);

        double resultado = numeros[posicion - 1];

        System.out.println("Resultado inicial: " + resultado);

        // Realizar cinco operaciones
        for (int i = 1; i <= 5; i++) {

            System.out.println("\nOperacion " + i);

            System.out.println("1. Sumar");
            System.out.println("2. Restar");
            System.out.println("3. Multiplicar");
            System.out.println("4. Dividir");

            resultado = calcularOperacion(resultado, auxiliar, operacion);

    // Generacion de letras con minimo cuatro vocales y cuatro consonantes

    static char[] generarLetras(int cantidad) {

        char[] letras = new char[cantidad];

        String vocales = "AEIOU";

        String consonantes = "BCDFGHJKLMNÑPQRSTVWXYZ";

        // Garantizar cuatro vocales
        for (int i = 0; i < 4; i++) {

            int posicion = aleatorio.nextInt(vocales.length());

            letras[i] = vocales.charAt(posicion);
        }

        // Garantizar cuatro consonantes
        for (int i = 4; i < 8; i++) {

            int posicion = aleatorio.nextInt(consonantes.length());

            letras[i] = consonantes.charAt(posicion);
        }

        // Completar las letras restantes aleatoriamente
        for (int i = 8; i < cantidad; i++) {

            if (aleatorio.nextBoolean()) {

                letras[i] = vocales.charAt(
                    aleatorio.nextInt(vocales.length())
                );

            } else {

                letras[i] = consonantes.charAt(
                    aleatorio.nextInt(consonantes.length())
                );
            }
        }

        // Mezclar las letras para que no aparezcan agrupadas
        for (int i = 0; i < letras.length; i++) {

            int posicion = aleatorio.nextInt(letras.length);

            char temporal = letras[i];

            letras[i] = letras[posicion];

            letras[posicion] = temporal;
        }

        return letras;
    }


    // Comprobar que una palabra puede formarse con las letras disponibles


static boolean puedeFormarse(String palabra, char[] letras) {

    String palabraNormalizada = normalizarPalabra(palabra);

    boolean[] utilizadas = new boolean[letras.length];

    if (palabraNormalizada.length() > letras.length) {
        return false;
    }

    for (int i = 0; i < palabraNormalizada.length(); i++) {

        char letra = palabraNormalizada.charAt(i);

        boolean encontrada = false;

        // Buscar una letra disponible que coincida
        for (int j = 0; j < letras.length && !encontrada; j++) {

            char letraDisponible = normalizarPalabra(
                String.valueOf(letras[j])
            ).charAt(0);

            if (letra == letraDisponible && !utilizadas[j]) {

                utilizadas[j] = true;

                encontrada = true;
            }
        }

        if (!encontrada) {
            return false;
        }
    }

    return true;
}



    // Normalizar palabras: mayusculas y vocales sin tilde

    static String normalizarPalabra(String palabra) {

        String texto = palabra.trim().toUpperCase();

        texto = Normalizer.normalize(
            texto,
            Normalizer.Form.NFD
        );

        texto = texto.replaceAll("\\p{M}", "");

        return texto;
    }

    // Realizar las operaciones de un jugador

    static double jugarCalculadora() {

        double numeroActual = generarNumero(1, 100);

        int fallos = 0;

        System.out.println("Numero inicial: " + numeroActual);

        long tiempoInicial = System.nanoTime();

        for (int i = 1; i <= cantidadOperaciones; i++) {

            int operacion = aleatorio.nextInt(4) + 1;

            int valor;

            if (operacion == 1 || operacion == 2) {

                valor = generarNumero(10, 99);

            } else {

                valor = generarNumero(1, 9);
            }

            double resultadoCorrecto = calcularOperacion(
                numeroActual,
                valor,
                operacion
            );

            System.out.println("\nOperacion " + i + " de "
                    + cantidadOperaciones);

            mostrarOperacion(operacion, numeroActual, valor);

            double respuesta = leerDouble("Introduce el resultado: ");

            if (Math.abs(respuesta - resultadoCorrecto) < 0.000001) {

                System.out.println("Correcto.");

            } else {

                System.out.println("Incorrecto.");

                System.out.println("El resultado correcto era: "
                        + resultadoCorrecto);

                fallos++;
            }

            // El resultado correcto pasa a ser el nuevo numero inicial
            numeroActual = resultadoCorrecto;
        }

        long tiempoFinal = System.nanoTime();

        double segundos = (tiempoFinal - tiempoInicial) / 1_000_000_000.0;

        double penalizacion = fallos * 20.0;

        double tiempoTotal = segundos + penalizacion;

        System.out.println("\nFallos: " + fallos);

        System.out.println("Penalizacion: " + penalizacion + " segundos");

        System.out.printf("Tiempo real: %.2f segundos%n", segundos);

        System.out.printf("Tiempo total: %.2f segundos%n", tiempoTotal);

        return tiempoTotal;
    }

        int[] ordenados = new int[6];

        }

        // Ordenacion mediante el metodo burbuja
        // Se utilizan dos bucles anidados

        for (int i = 0; i < ordenados.length - 1; i++) {

            for (int j = 0; j < ordenados.length - 1 - i; j++) {

                if (ordenados[j] > ordenados[j + 1]) {

                    int temporal = ordenados[j];

                    ordenados[j] = ordenados[j + 1];

                    ordenados[j + 1] = temporal;

        }

        System.out.println("\n\nTurno de " + jugador1);

        int[] respuestaJ1 = introducirOrden();

        System.out.println("\nTurno de " + jugador2);

        int[] respuestaJ2 = introducirOrden();

        int aciertosJ2 = contarAciertos(respuestaJ2, ordenados);

        mostrarArray(respuestaJ1);

        System.out.println("Aciertos de " + jugador1 + ": " + aciertosJ1);

        mostrarArray(respuestaJ2);

        System.out.println("Aciertos de " + jugador2 + ": " + aciertosJ2);

        System.out.println("Orden correcto:");

        mostrarArray(ordenados);

    // Contar las posiciones correctas

    static int contarAciertos(int[] respuesta, int[] solucion) {

        int aciertos = 0;

        for (int i = 0; i < respuesta.length; i++) {

            if (respuesta[i] == solucion[i]) {

                aciertos++;
            }
        }

        return aciertos;
    }

    // Calcular una operacion matematica sin entrada ni salida


static double calcularOperacion(
        double numero1,
        double numero2,
        int operacion) {

    double resultado = 0;

    if (operacion == 1) {

        resultado = numero1 + numero2;

    } else if (operacion == 2) {

        resultado = numero1 - numero2;

    } else if (operacion == 3) {

        resultado = numero1 * numero2;

    } else if (operacion == 4) {

        if (numero2 != 0) {

            resultado = numero1 / numero2;

        } else {

            throw new ArithmeticException("Division entre cero.");
        }
    }

    return resultado;
}



    // Generar un numero aleatorio entre dos valores incluidos

    static int generarNumero(int minimo, int maximo) {

        return aleatorio.nextInt(maximo - minimo + 1) + minimo;
    }

    // =====================================================
    // 14. FUNCIONES DE ENTRADA Y SALIDA
    // =====================================================

    // Leer un numero entero controlando errores

    static int leerEntero(String mensaje) {

        while (true) {

            System.out.print(mensaje);

            String entrada = teclado.nextLine().trim();

            try {

                return Integer.parseInt(entrada);

            } catch (NumberFormatException e) {

                System.out.println("Error: debes introducir un numero entero.");
            }
        }
    }


    // Leer un numero decimal controlando errores

    static double leerDouble(String mensaje) {

        while (true) {

            System.out.print(mensaje);

            String entrada = teclado.nextLine().trim();

            entrada = entrada.replace(',', '.');

            try {

                double numero = Double.parseDouble(entrada);

                if (Double.isFinite(numero)) {
                    return numero;
                }

                System.out.println("Introduce un numero valido.");

            } catch (NumberFormatException e) {

                System.out.println("Error: introduce un numero valido.");
            }
        }
    }


    // Leer una cadena de texto

    static String leerTexto(String mensaje) {

        String texto;

        do {

            System.out.print(mensaje);

            texto = teclado.nextLine().trim();

            if (texto.isEmpty()) {

                System.out.println("No puedes dejar la respuesta vacia.");
            }

        } while (texto.isEmpty());

        return texto;
    }


    // Confirmar si una palabra existe en español

    static boolean confirmarPalabra() {

        int opcion;

        do {

            System.out.println("1. La palabra existe en español.");
            System.out.println("2. La palabra no existe.");

            opcion = leerEntero("Selecciona una opcion: ");

            if (opcion != 1 && opcion != 2) {

                System.out.println("Opcion incorrecta.");
            }

        } while (opcion != 1 && opcion != 2);

        return opcion == 1;
    }
